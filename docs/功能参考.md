# 功能参考

## WebSocket 会话交互（实时）

进入运行中的任务会话，发送消息让 agent 继续执行（需在 asyncio 中）：

```python
import asyncio
from monkeycode import MonkeyCodeClient

async def main():
    client = MonkeyCodeClient(email="you@example.com", password="your_password")
    client.login()
    # 发送消息并等待 agent 开始执行
    events = await client.send_task_message(task_id, "请把 README 改成 3 步")
    for e in events:
        print(e.type, e.data)
    # 或长连接实时订阅
    async with client.open_task_stream(task_id, mode="new") as stream:
        await stream.send_user_input("继续")
        while True:
            chunk = await stream.recv(timeout=60)
            if chunk.type == "task-ended":
                break

asyncio.run(main())
```

流控制（长任务）：

```python
async with client.open_task_stream(task_id, mode="new") as stream:
    await stream.send_reply_question("request_id", ["答案"])
```

## 任务控制 / 命令选项 / 终端

切换运行中任务的模型（走 control 通道，需在 asyncio 中）：

```python
import asyncio
from monkeycode import MonkeyCodeClient

async def main():
    client = MonkeyCodeClient(email="you@example.com", password="your_password")
    client.login()
    result = await client.switch_task_model(task_id, model_id)
    print(result)   # {"success": true, "session_id": "...", "model": {...}}

    # 或手动打开控制通道做其他同步调用（restart / 端口转发等）
    async with client.open_task_control(task_id) as stream:
        await stream.send_call("port_forward_list", {})
        chunk = await stream.recv(timeout=30)
        print(chunk.data)

asyncio.run(main())
```

虚拟机终端实时交互（WebSocket，需在 asyncio 中）：

```python
async def terminal_demo():
    client = MonkeyCodeClient(email="you@example.com", password="your_password")
    client.login()
    async with client.open_terminal(vm_id, col=80, row=24) as term:
        await term.send_data("ls -la\r")
        while True:
            chunk = await term.recv(timeout=30)
            print(chunk.data)          # 终端输出（已 base64 解码）
            if chunk.type == "error":
                break

asyncio.run(terminal_demo())
```
